    let currentIndex = 0;
    let answeredCount = 0;
    let score = 0;

    function currentCard() {
      return vocabList[currentIndex];
    }

    function renderCard() {
      const item = currentCard();
      document.getElementById("cardCounter").textContent = `${currentIndex + 1} / ${vocabList.length}`;
      document.getElementById("frontPos").textContent = item.pos;
      document.getElementById("frontWord").textContent = item.word;
      document.getElementById("frontIpa").textContent = item.ipa;
      document.getElementById("backMeaningVi").textContent = item.meaningVi;
      document.getElementById("backDefEn").textContent = item.defEn;
      document.getElementById("backExample").textContent = `"${item.example}"`;
      document.getElementById("flashcard").classList.remove("flipped");
    }

    function flipCurrentCard(e) {
      if (e) e.stopPropagation();
      document.getElementById("flashcard").classList.toggle("flipped");
    }

    function handleCardClick(e) {
      if (e.target.closest("button")) return;
      flipCurrentCard();
    }

    function nextCard() {
      currentIndex = (currentIndex + 1) % vocabList.length;
      renderCard();
    }

    function prevCard() {
      currentIndex = (currentIndex - 1 + vocabList.length) % vocabList.length;
      renderCard();
    }

    function speakText(text, e) {
      if (e) e.stopPropagation();
      if (!("speechSynthesis" in window)) {
        alert("Browser does not support speech synthesis.");
        return;
      }
      window.speechSynthesis.cancel();
      const utter = new SpeechSynthesisUtterance(text);
      utter.lang = "en-US";
      utter.rate = 0.95;
      window.speechSynthesis.speak(utter);
    }

    function toggleListView() {
      const listEl = document.getElementById("allWordsList");
      const isHidden = listEl.style.display === "none";
      if (isHidden) {
        listEl.innerHTML = vocabList.map((v, i) => `
          <div class="list-item">
            <div style="font-size:11px; font-weight:700; color:#0284c7;">#${i + 1} • ${v.pos}</div>
            <h4>${v.word}</h4>
            <div class="ipa-small">${v.ipa} — <strong>${v.meaningVi}</strong></div>
            <p style="font-size:13.5px; color:#475569; margin-bottom:10px;">${v.defEn}</p>
            <button class="btn" style="padding:6px 12px; font-size:12px;" onclick="speakText('${v.word.replace(/'/g, "\\'")}')">🔊 Listen</button>
          </div>
        `).join("");
        listEl.style.display = "grid";
      } else {
        listEl.style.display = "none";
      }
    }

    function renderQuiz() {
      const container = document.getElementById("quizContainer");
      answeredCount = 0;
      score = 0;
      document.getElementById("scoreBox").style.display = "none";

      container.innerHTML = quizData.map((item, qIdx) => `
        <div class="quiz-card" id="qcard-${qIdx}">
          <div class="q-num">Question ${qIdx + 1} of ${quizData.length}</div>
          <div class="q-text">${item.q}</div>
          <div class="options">
            ${item.options.map((opt, oIdx) => `
              <button class="option-btn" onclick="selectOption(${qIdx}, ${oIdx}, this)">${opt}</button>
            `).join("")}
          </div>
          <div class="explanation" id="exp-${qIdx}">${item.explain}</div>
        </div>
      `).join("");
    }

    function selectOption(qIdx, oIdx, btn) {
      const card = document.getElementById(`qcard-${qIdx}`);
      const buttons = card.querySelectorAll(".option-btn");
      if (buttons[0].disabled) return;

      buttons.forEach(b => b.disabled = true);
      const correctIdx = quizData[qIdx].correct;

      if (oIdx === correctIdx) {
        btn.classList.add("correct");
        score++;
      } else {
        btn.classList.add("wrong");
        buttons[correctIdx].classList.add("correct");
      }

      document.getElementById(`exp-${qIdx}`).classList.add("show");
      answeredCount++;

      if (answeredCount === quizData.length) {
        showScore();
      }
    }

    function showScore() {
      const box = document.getElementById("scoreBox");
      const val = document.getElementById("scoreValue");
      const fb = document.getElementById("scoreFeedback");
      val.textContent = `${score} / ${quizData.length}`;

      if (score >= 7) {
        fb.textContent = "Excellent command of C1 vocabulary and digital dependency concepts!";
      } else if (score >= 5) {
        fb.textContent = "Good effort! Review the flashcards and explanations to master the remaining terms.";
      } else {
        fb.textContent = "Keep practicing with the flashcards and vocal audio, then try the quiz again.";
      }
      box.style.display = "block";
      box.scrollIntoView({ behavior: "smooth", block: "nearest" });
    }

    function resetQuiz() {
      renderQuiz();
    }

    function switchTab(tabName, btn) {
      document.querySelectorAll(".section").forEach(s => s.classList.remove("active"));
      document.querySelectorAll(".tab-btn").forEach(b => b.classList.remove("active"));
      document.getElementById(tabName).classList.add("active");
      btn.classList.add("active");
    }
